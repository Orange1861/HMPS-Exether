cast_arcana_id_gryphon_is_valid = {

    custom_tooltip = {
        text = constellation_nodes_connected_tt
        OR = {
            AND = {
                scope:arcana_id_gryphon ?= 4
                OR = {
                    has_modifier = country_arcana_gryphon_3
                    has_modifier = country_arcana_gryphon_6
                }
            }

            AND = {
                scope:arcana_id_gryphon ?= 6
                OR = {
                    has_modifier = country_arcana_gryphon_1
                    has_modifier = country_arcana_gryphon_2
                    has_modifier = country_arcana_gryphon_4
                    has_modifier = country_arcana_gryphon_5
                }
            }


            AND = {
                NOT = { scope:arcana_id_gryphon ?= 4 }
                NOT = { scope:arcana_id_gryphon ?= 6 }
            }
        }
    }

    religion = {
        has_discrimination_trait = gryphon_domain
    }
    NOT = { 
        has_modifier = country_arcana_card_cooldown
    }
    root = {
        custom_tooltip = {
            text = has_reserves_tt
            root.arcana >= arcana_cost
        }
    }
}
