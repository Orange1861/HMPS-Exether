je_shattered_ring = {
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	
	group = je_group_historical_content

	scripted_progress_bar = orc_ring_bad_progress_bar


	immediate = {
		c:KHD ?= {
			save_scope_as = khd_scope
		}

		c:TSH ?= {
			save_scope_as = tsh_scope
		}
		c:HOZ ?= {
			save_scope_as = hoz_scope
		}
	}

		status_desc = {
		first_valid = {
			triggered_desc = {
				desc = ring_pipe_dream_loc
				trigger = {
					scope:journal_entry = {
						"scripted_bar_progress(orc_ring_bad_progress_bar)" <= 1100
					}
				}
			}
			triggered_desc = {
				desc = ring_getting_serious_loc
				trigger = {
					scope:journal_entry = {
						"scripted_bar_progress(orc_ring_bad_progress_bar)" <= 1600
					}
				}
			}
			triggered_desc = {
				desc = ring_nearing_ratification_loc
				trigger = {
					scope:journal_entry = {
						"scripted_bar_progress(orc_ring_bad_progress_bar)" > 1600
					}
				}
			}
		}
	}

	complete = {
		OR = {
			custom_tooltip = {
				text = stihug_power_bloc_tt
				power_bloc ?= {
					any_power_bloc_member = {
						c:TSH ?= this
					}
				}
			}
			
			c:TSH ?= {
				is_subject_of = root
			}
			custom_tooltip = {
					text = kill_khd_tt
					NOT = {
						exists = c:KHD
					}
				}
			custom_tooltip = {
				text = kill_stihug_tt
				NOT = {
					exists = c:TSH
				}
			}
				
				
			
			
		}
	}
	fail = {
		OR = {
			custom_tooltip = {
				text = khd_power_bloc_tt
				c:KHD ?= {
					power_bloc ?= {
						any_power_bloc_member = {
							c:TSH ?= this
						}
					}
				}	
			}	
		}
			

	}

	on_fail = {
		trigger_event = rust_gorge.101
	}

	on_complete = {
		trigger_event = rust_gorge.100
	}

	invalid = { # If khaduras and Stihugnado go to war, it's over
		OR = {
			scope:journal_entry = {
				"scripted_bar_progress(orc_ring_bad_progress_bar)" <= 0
			}
			c:TSH = {
				any_enemy_in_war = {
					c:KHD ?= this
				}
			}
			custom_tooltip = { #if an interloper pulls stihugnado into the bloc, it's over
				text = khd_power_bloc_tt2
				c:TSH ?= {
					is_in_power_bloc = yes
					NOT = { is_in_same_power_bloc = c:KHD }
					NOT = { is_in_same_power_bloc = root }
				}
			}
		}
		
	}

	on_invalid = {
		trigger_event = rust_gorge.105
	}

	should_be_pinned_by_default = yes

	weight = 1000
}

je_ring_on_it = {
	icon = "gfx/interface/icons/event_icons/event_centralization.dds"
	
	group = je_group_historical_content

	scripted_progress_bar = orc_ring_good_progress_bar


	immediate = {
		c:KHD ?= {
			save_scope_as = khd_scope
		}

		c:TSH ?= {
			save_scope_as = tsh_scope
		}
		c:HOZ ?= {
			save_scope_as = hoz_scope
		}
	}

		status_desc = {
		first_valid = {
			triggered_desc = {
				desc = ring_pipe_dream_loc
				trigger = {
					scope:journal_entry = {
						"scripted_bar_progress(orc_ring_good_progress_bar)" <= 1100
					}
				}
			}
			triggered_desc = {
				desc = ring_getting_serious_loc
				trigger = {
					scope:journal_entry = {
						"scripted_bar_progress(orc_ring_good_progress_bar)" <= 1600
					}
				}
			}
			triggered_desc = {
				desc = ring_nearing_ratification_loc
				trigger = {
					scope:journal_entry = {
						"scripted_bar_progress(orc_ring_good_progress_bar)" > 1600
					}
				}
			}
		}
	}

	on_weekly_pulse = {
		if = {
			limit = {
				scope:journal_entry = {
					"scripted_bar_progress(orc_ring_good_progress_bar)" >= 2000
				}
			}
			set_variable = ring_of_steel_var
		}
	}
	complete = {
		custom_tooltip = {
			text = stihug_power_bloc_tt
			power_bloc ?= {
				any_power_bloc_member = {
					c:TSH ?= this
				}
			}
		}	
	}
	fail = {

		OR = {
			custom_tooltip = {
				text = tsh_subject_tt
				c:TSH ?= {
				is_subject = yes
					top_overlord ?= {
						NOT = {
							 this = root
						}
					}
				}
			}
			
			custom_tooltip = {
				text = khd_power_bloc_tt2
				c:TSH ?= {
					is_in_power_bloc = yes
					NOT = { is_in_same_power_bloc = c:KHD }
				}
			}
			AND = {
				custom_tooltip = {
					text = kill_stihug_tt
					NOT = {
						exists = c:TSH
					}
				}
			}
			
		}
			

	}

	on_fail = {
		trigger_event = rust_gorge.103
	}

	on_complete = {
		trigger_event = rust_gorge.102
	}

	invalid = { # If khaduras and Stihugnado go to war, it's over
		OR = {
			scope:journal_entry = {
				"scripted_bar_progress(orc_ring_good_progress_bar)" <= 0
			}
			c:TSH = {
				any_enemy_in_war = {
					c:KHD ?= this
				}
			}
		}	
	}

	on_invalid = {
		trigger_event = rust_gorge.106
	}

	should_be_pinned_by_default = yes

	weight = 1000
}
