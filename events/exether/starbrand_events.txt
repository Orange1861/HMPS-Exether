starbrand_events.1 = { #mines displace monks
	type = country_event
	title = starbrand_events.1.t
	desc = starbrand_events.1.d
	flavor = starbrand_events.1.f

	duration = 3
    dlc = dlc_roe
    event_image = {
        video = "rock_elementals"
    }

	placement = scope:monk_state
	cooldown = { years = 55 }

	trigger = {
		OR = {
			ig:ig_devout ?= {
					is_marginal = no
				}
			ig:ig_mages ?= {
					is_marginal = no
				}	
		}
		ig:ig_industrialists ?= {
			is_marginal = no
		}	
		
		any_scope_state = {
			is_incorporated = yes
			state_has_building_group_levels = {
				type  = bg_mining value >= 5
			}
		}
		
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	immediate = {
		random_scope_state = {
			limit = {
				is_incorporated = yes
				state_has_building_group_levels = {
					type  = bg_mining value >= 5
				}
			}
			save_scope_as = monk_state
		}
	}


	option = {
		name = starbrand_events.1.a #why do we care?
		default_option = yes
		ig:ig_industrialists ?= {
			add_modifier = {
				name = ignored_starbrand_protests
				years = 10
				is_decaying = yes
			}
		}			
				scope:monk_state = {
					add_modifier = {
						name = mine_protests
						years = 10
					}
			}
		
	}

	option = {
		name = starbrand_events.1.b #send in the military!
		if = {
			limit = {
				ig:ig_devout ?= {
					is_marginal = no
				}
			}
			ig:ig_devout ?= {
				add_modifier = {
					name = crushed_monastic_protest
					years = 10
					is_decaying = yes
				}
			}
		}

		if = {
			limit = {
				ig:ig_mages ?= {
					is_marginal = no
				}
			}
			ig:ig_mages ?= {
				add_modifier = {
					name = crushed_monastic_protest
					years = 10
					is_decaying = yes
				}
			}
		}
		scope:monk_state = {
			state_region = {
				start_harvest_condition = rock_elementals
