locari_regional.1 = { #vagrant hiadud musicians
	type = country_event
	title = locari_regional.1.t
	desc = locari_regional.1.d
	flavor = locari_regional.1.f

	duration = 3
    dlc = dlc_roe
    event_image = {
        video = "rp1_social_banditry"
    }

	placement = scope:vagrant_state
	cooldown = { years = 40 }

	trigger = {
		any_scope_state = {
			is_incorporated = yes
			state_is_in_locari = yes
		}
		
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	immediate = {
		random_scope_state = {
			limit = {
				is_incorporated = yes
				state_is_in_locari = yes
			}
			save_scope_as = vagrant_state
		}
	}


	option = {
		name = locari_regional.1.a #why do we care?
		default_option = yes
			scope:vagrant_state = {
				add_radicals_in_state = {
					value = 0.15
				}
				if = {
					limit = {
						any_neighbouring_state = {
							owner = root
						}
					}
					random_neighbouring_state = {
						limit = {
							owner = root
						}
						add_radicals_in_state = {
							value = 0.15
						}
					}
				}
			}
	}

	option = {
		name = locari_regional.1.b #send in the military!
		ig:ig_armed_forces ?= {
			add_modifier = {
				name = menial_tasks
				years = 10
				is_decaying = yes
			}
		}
		scope:vagrant_state = {
			add_radicals_in_state = {
				value = 0.05
			}
		}

	}
	option = {
		name = locari_regional.1.c #can't the adventurers handle it?
		highlighted_option = yes
		trigger = {
			custom_tooltip = {
				text = adventurer_coverage_positive_tt
				scope:vagrant_state = {
					state_adventurer_coverage_balance > 0
				}
			}
		}
		scope:vagrant_state = {
			add_radicals_in_state = {
				value = 0.05
			}
		}
	}
	option = {
		name = locari_regional.1.e #BAN DANCING
		highlighted_option = yes
		trigger = {
			NOT = {
				has_law = law_type:law_no_police
			}
		}
		scope:vagrant_state = {
			add_radicals_in_state = {
				value = 0.05
			}
		}
	}
	option = {
		name = locari_regional.1.g #MUSIC DUEEEEEEEEEEEEEEEEEELLLLLLLLLLLLLLLLLLLLLLLL
		highlighted_option = yes
		trigger = {
			any_primary_culture = {
				has_discrimination_trait = hiadud_heritage
			}
		}
		random_list = {
			50 = {
				add_modifier = {

				}
			}
			50 = {
				add_modifier = {
					name = lost_music_duel
					years = 10
				}
			}
		}
	}
}

locari_regional.2 = { #refugees escaping the naga

		any_scope_state = {

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

			limit = {
				is_incorporated = yes

					owner = {
						OR = {
							country_has_primary_culture = cu:gulf_naga

						}
					}
				}
			}
			save_scope_as = refugee_state
			random_neighbouring_state = {
				limit = {
					owner = {
						OR = {
							country_has_primary_culture = cu:gulf_naga

						}
					}
				}
				owner = {
					save_scope_as = escaped_from_country

			}

			limit = {
				ig:ig_petty_bourgeoisie ?= {
					is_marginal = no
				}
			}
			ig:ig_petty_bourgeoisie ?= {
				add_modifier = {
					name = accepted_refugees
					years = 5
				}
			}
		}
	}

			name = turned_back_refugees
			years = 10
