adventurer_coverage.1 = { #adventurer scout program!
	type = country_event
	title = adventurer_coverage.1.t
	desc = adventurer_coverage.1.d
	flavor = adventurer_coverage.1.f

	duration = 3
    dlc = dlc_roe
    event_image = {
        video = "southamerica_child_labor"
    }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	cooldown = { years = 65 }

	trigger = {
		any_scope_state = {
			is_incorporated = yes
			state_adventurer_coverage_balance <= -35
		}
		NOT = { has_variable = adventurer_cooldown_variable } #only one adventurer event every 6 years
	}

	immediate = {
		set_variable = {
			name = adventurer_cooldown_variable
			years = 6
		}
	}


	option = {
		name = adventurer_coverage.1.a #the children yearn for the forests!
		add_modifier = {
				name = child_adventurers
				years = 10
			}
	}

	option = {
		name = adventurer_coverage.1.b #we should take measures to prevent this
		default_option = yes
	}

	option = {
		name = adventurer_coverage.1.c #this is child labor!
		trigger = {
			has_law = law_type:law_restricted_child_labor
		}
        add_modifier = {
			name = wards_of_the_state
			years = 10
		}
		add_radicals = {
			pop_type = adventurers
			value = 0.03
		}
		add_radicals = {
			pop_type = wizards
			value = 0.03
		}
	}
}
